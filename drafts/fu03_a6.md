```python
# Follow-up 3, the section 11 reading of its three precision contrasts (section 10 row of 2026-11-27, label
# fu03_a6), computed from frozen outputs after every result was read.
FU03_A6 = f'{AN}/fu03_a6_20261127'
ROLE_FU03_A6 = ('follow-up 3 (outside the pre-registered study): the section 11 reading that its row of 2026-10-16 '
                'fixed, computed on 2026-11-27 after every result was read; post hoc rows labelled')
FU03_A6_ROWS = [('P1', 'v6e highest vs A100 highest minus recompile v6e (plan)'),
                ('P2', 'A100 highest vs CPU minus recompile A100 (plan)'),
                ('Ha2', 'A100 highest vs CPU minus recompile A100, seeds 1 and 2 (post hoc)'),
                ('Hd1', 'v6e vs A100 minus recompile v6e, seeds 1 and 2 (post hoc)'),
                ('Hd3', 'A100 vs CPU minus recompile A100 (post hoc)'),
                ('Hc1', 'Precision, A100 minus recompile A100 (post hoc)')]
FU03_A6_READ = {'attributed to the platform': 'platform', 'attributed to the default matmul precision': 'precision',
                'a difference beyond recompilation': 'beyond', 'no difference beyond recompilation shown': 'not beyond',
                'not distinguished from recompilation': 'not beyond'}
FU03_A6_TEXT = {('P1', 'flip_top1'): 'platform', ('P1', 'flip_success'): 'platform', ('P1', 'flip_matters'): 'precision',
                ('P2', 'flip_top1'): 'precision', ('P2', 'flip_success'): 'precision',
                ('P2', 'flip_matters'): 'precision', ('Ha2', 'flip_top1'): 'platform'}


def table_fu03_s11(out, main_dir):
  """Supplementary table: the highest-precision contrasts of follow-up 3 minus RC of their first platform (plan), and
  four post hoc comparisons. The guards hold the readings the Status text states."""
  res = read(main_dir, f'{FU03_A6}/results.csv')
  chk = read(main_dir, f'{FU03_A6}/checks.csv')
  if not chk.value.astype(str).str.startswith('pass').all():
    raise SystemExit(f'{FU03_A6}: a check did not pass')
  got = {(r.id, r.statistic): FU03_A6_READ[r.reading] for r in res.itertuples() if (r.id, r.statistic) in FU03_A6_TEXT}
  if got != FU03_A6_TEXT:
    raise SystemExit(f'{FU03_A6}: readings changed ({got}); the text states {FU03_A6_TEXT}')
  rows, plain = [], []
  for rid, lab in FU03_A6_ROWS:
    g = res[res.id == rid].set_index('statistic')
    vals = {l: diff_ci(g.loc[k]) for k, l in STAT3}
    rd = {l: FU03_A6_READ[g.loc[k, 'reading']] for k, l in STAT3}
    rows += [{'Comparison': lab, **vals}, {'Comparison': '\\INDENT reading', **rd}]
    plain += [{'Comparison': lab, 'Line': 'difference (points)', **vals}, {'Comparison': lab, 'Line': 'reading', **rd}]
  tdf = pd.DataFrame(rows, columns=['Comparison'] + [l for _, l in STAT3])
  write_table(out, 'TS_fu03_s11', pd.DataFrame(plain),
              'Follow-up 3: highest-precision contrasts against recompilation (plan section 11)', ROLE_FU03_A6,
              notes=['Differences in points, with 95% intervals over targets. The highest-precision arms ran seeds 1 '
                     'and 2. Recompilation uses seeds 1 to 5 unless a row names seeds 1 and 2. Plan rows use the '
                     'reference and the test that the plan fixes. Platform: the interval lies above 0. Precision: the '
                     'interval contains 0, and the rule attributes the rate to the default matrix-product precision. '
                     'Beyond and not beyond apply the same test to comparisons that the rule does not read. '
                     'Recompilation of the CPU was not measured.'],
              colspec='llll', tex_df=tdf)
```
